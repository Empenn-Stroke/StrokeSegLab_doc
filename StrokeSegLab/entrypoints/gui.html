301        entry.insert(0, str(self.threshold_var.get()))
302        entry.pack()
303        
304        # Synchronize Entry with Scale
305        def on_entry_change(*args):
306            try:
307                value = float(entry.get())
308                if 0.0 <= value <= 1.0:
309                    self.threshold_var.set(round(value, 2))
310            except ValueError:
311                pass
312        
313        # Synchronize Scale with Entry
314        def on_scale_change(value):
315            entry.delete(0, tk.END)
316            entry.insert(0, str(round(float(value), 2)))
317
318        entry.bind("<KeyRelease>", lambda event: on_entry_change())
319        scale.config(command=on_scale_change)
320
321        ok_button = tk.Button(threshold_window, text="OK", command=threshold_window.destroy)
322        ok_button.pack(pady=10)

590    def _predict(self)->None:
591        """
592        Run the prediction pipeline for all subjects.
593
594        Depending on the model:
595        - 2 channels (FLAIR and T1), 't1' is the T1 image and 'flair' is the FLAIR image.
596        - 1 channel, 't1' contains either the T1 or the FLAIR image, and 'flair' is None.
597
598        It creates a temporary directory and then, for each subject : 
599        - Run preprocessing, inference and postprocessing
600        - Checks periodically if a stop is requested
601        - Update the UI during the process using self.window.after because Tkinter isn't thread safe, we can't modify the UI from a secondary thread. So 'after' is scheduling an update in the main thread
602        """
603        threshold = self.threshold_var.get()
604        len_nii_paths= len(self.nii_paths)
605        i=0
606        for t1, flair in self.nii_paths.items():
607            temp_dir = tempfile.mkdtemp(prefix="unet_preprocess")
608            try:
609                if self.check_stop():
610                    raise InterruptedError("Action was cancelled by the user.")
611                if flair is None:
612                    s = f"Working on: {os.path.basename(t1)} ({i+1}/{len_nii_paths})"
613                    self.logger.info(f"Starting processing on: {os.path.basename(t1)}")
614                else :
615                    s=f"Starting processing on: ({os.path.basename(t1)},{os.path.basename(flair)})"
616                    self.logger.info(f"Starting processing on: ({os.path.basename(t1)},{os.path.basename(flair)})")
617                self.window.after(0,self._update_stringvar,self.working_on_text,s) # We schedule a method that will be executed in the main thread
618                data, affine, bbox,original_shape, trsf_path, old_spacing, padding, bet, MNI_base_image  = self.preprocessor.run(t1,flair,temp_dir)
619                if self.check_stop():
620                    raise InterruptedError("Action was cancelled by the user.")
621                data = self.inference.run(data)
622                if self.check_stop():
623                    raise InterruptedError("Action was cancelled by the user.")
624                if self.open_viewer.get() and i==0: # Only open the viewer for the first image
625                    self.postprocessor.run(data,affine,t1,bbox,original_shape,temp_dir,trsf_path,old_spacing,padding,bet,MNI_base_image,threshold,True)
626                else : 
627                    self.postprocessor.run(data,affine,t1,bbox,original_shape,temp_dir,trsf_path,old_spacing,padding,bet,MNI_base_image,threshold)
628                i+=1
629                self.preprocessor.clean(temp_dir)
630            except Exception as e: # If an exception occurs, processing of the current subject is stopped, a message is logged to inform the user, and the process continues with the next subject
631                self.logger.error(f"Erreur lors du traitement de {t1} : {e}")
632                self.success = False
633                self.preprocessor.clean(temp_dir)
634        self.window.after(0,self._update_result) # Call for update_result in the main thread

591    def _predict(self)->None:
592        """
593        Run the prediction pipeline for all subjects.
594
595        Depending on the model:
596        - 2 channels (FLAIR and T1), 't1' is the T1 image and 'flair' is the FLAIR image.
597        - 1 channel, 't1' contains either the T1 or the FLAIR image, and 'flair' is None.
598
599        It creates a temporary directory and then, for each subject : 
600        - Run preprocessing, inference and postprocessing
601        - Checks periodically if a stop is requested
602        - Update the UI during the process using self.window.after because Tkinter isn't thread safe, we can't modify the UI from a secondary thread. So 'after' is scheduling an update in the main thread
603        """
604        threshold = self.threshold_var.get()
605        len_nii_paths= len(self.nii_paths)
606        i=0
607        for t1, flair in self.nii_paths.items():
608            temp_dir = tempfile.mkdtemp(prefix="unet_preprocess")
609            try:
610                if self.check_stop():
611                    raise InterruptedError("Action was cancelled by the user.")
612                if flair is None:
613                    s = f"Working on: {os.path.basename(t1)} ({i+1}/{len_nii_paths})"
614                    self.logger.info(f"Starting processing on: {os.path.basename(t1)}")
615                else :
616                    s=f"Starting processing on: ({os.path.basename(t1)},{os.path.basename(flair)})"
617                    self.logger.info(f"Starting processing on: ({os.path.basename(t1)},{os.path.basename(flair)})")
618                self.window.after(0,self._update_stringvar,self.working_on_text,s) # We schedule a method that will be executed in the main thread
619                data, affine, bbox,original_shape, trsf_path, old_spacing, padding, bet, MNI_base_image  = self.preprocessor.run(t1,flair,temp_dir)
620                if self.check_stop():
621                    raise InterruptedError("Action was cancelled by the user.")
622                data = self.inference.run(data)
623                if self.check_stop():
624                    raise InterruptedError("Action was cancelled by the user.")
625                if self.open_viewer.get() and i==0: # Only open the viewer for the first image
626                    self.postprocessor.run(data,affine,t1,bbox,original_shape,temp_dir,trsf_path,old_spacing,padding,bet,MNI_base_image,threshold,True)
627                else : 
628                    self.postprocessor.run(data,affine,t1,bbox,original_shape,temp_dir,trsf_path,old_spacing,padding,bet,MNI_base_image,threshold)
629                i+=1
630                self.preprocessor.clean(temp_dir)
631            except Exception as e: # If an exception occurs, processing of the current subject is stopped, a message is logged to inform the user, and the process continues with the next subject
632                self.logger.error(f"Erreur lors du traitement de {t1} : {e}")
633                self.success = False
634                self.preprocessor.clean(temp_dir)
635        self.window.after(0,self._update_result) # Call for update_result in the main thread
